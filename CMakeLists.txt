cmake_minimum_required(VERSION 3.24)
project(neo_fft VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(NEO_FFT_BUILD_VAPOURSYNTH "Build the VS plugin" ON)
option(NEO_FFT_ENABLE_HIGHWAY "Build Highway spectral kernels" ON)
include(cmake/Dependencies.cmake)
add_library(neo_fft_core STATIC src/spectral/fft.cpp src/algorithms/windows.cpp
  src/algorithms/plan.cpp src/kernels/scalar/spectral.cpp)
target_compile_definitions(neo_fft_core PRIVATE NEO_FFT_ENABLE_HIGHWAY=$<BOOL:${NEO_FFT_ENABLE_HIGHWAY}>)
target_compile_features(neo_fft_core PUBLIC cxx_std_17)
target_include_directories(neo_fft_core PUBLIC src)
target_include_directories(neo_fft_core SYSTEM PUBLIC "${dualsynth2_SOURCE_DIR}/include")
target_include_directories(neo_fft_core SYSTEM PRIVATE "${pocketfft_SOURCE_DIR}")
# Fixed baseline ISA, synchronous execution, no process-wide FFT plan cache.
target_compile_definitions(neo_fft_core PRIVATE POCKETFFT_NO_MULTITHREADING POCKETFFT_CACHE_SIZE=0)
if(MSVC)
  target_compile_options(neo_fft_core PUBLIC /fp:strict PRIVATE /W4)
else()
  target_compile_options(neo_fft_core PUBLIC -fno-fast-math -ffp-contract=off PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(neo_fft_core PROPERTIES POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(NEO_FFT_ENABLE_HIGHWAY)
  foreach(option TESTS EXAMPLES CONTRIB INSTALL)
    set(HWY_ENABLE_${option} OFF CACHE BOOL "" FORCE)
  endforeach()
  set(HWY_FORCE_STATIC_LIBS ON CACHE BOOL "" FORCE)
  FetchContent_Declare(highway
    URL https://github.com/google/highway/archive/refs/tags/1.4.0.tar.gz
    URL_HASH SHA256=e72241ac9524bb653ae52ced768b508045d4438726a303f10181a38f764a453c
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(highway)
  target_sources(neo_fft_core PRIVATE src/kernels/highway/spectral.cpp)
  target_link_libraries(neo_fft_core PUBLIC hwy)
  target_compile_definitions(neo_fft_core PRIVATE HWY_COMPILE_ALL_ATTAINABLE)
  set_target_properties(hwy PROPERTIES POSITION_INDEPENDENT_CODE ON)
  option(NEO_FFT_HWY_SCALAR_ONLY "Compile only the portable Highway scalar target" OFF)
  if(NEO_FFT_HWY_SCALAR_ONLY)
    target_compile_definitions(neo_fft_core PRIVATE HWY_COMPILE_ONLY_SCALAR)
  endif()
endif()

include(CTest)
if(NEO_FFT_BUILD_VAPOURSYNTH)
  include(cmake/VapourSynth.cmake)
  add_library(neo_fft MODULE src/plugin/vapoursynth.cpp)
  target_link_libraries(neo_fft PRIVATE neo_fft_core DualSynth::dualsynth)
  target_include_directories(neo_fft SYSTEM PRIVATE "${NEO_FFT_VS_INCLUDE}")
  set_target_properties(neo_fft PROPERTIES PREFIX "" OUTPUT_NAME "neo-fft"
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  if(MINGW)
    target_link_options(neo_fft PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  if(BUILD_TESTING)
    option(NEO_FFT_TEST_VAPOURSYNTH "Run admission tests with an installed Python/VS runtime" ON)
    if(NEO_FFT_TEST_VAPOURSYNTH)
      find_package(Python3 REQUIRED COMPONENTS Interpreter)
      add_test(NAME vs.admission COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/vapoursynth/admission.py" --plugin "$<TARGET_FILE:neo_fft>")
      set_tests_properties(vs.admission PROPERTIES RUN_SERIAL TRUE TIMEOUT 120 FAIL_REGULAR_EXPRESSION "Core freed but")
      add_test(NAME harness.protocol COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/differential/test_protocol.py")
    endif()
  endif()
endif()
if(BUILD_TESTING)
  add_executable(neo_fft_foundation_tests tests/unit/foundation.cpp)
  target_link_libraries(neo_fft_foundation_tests PRIVATE neo_fft_core)
  add_test(NAME core.foundation COMMAND neo_fft_foundation_tests)
  add_executable(neo_fft_algorithm_tests tests/unit/algorithms.cpp)
  target_link_libraries(neo_fft_algorithm_tests PRIVATE neo_fft_core)
  add_test(NAME core.algorithms COMMAND neo_fft_algorithm_tests)
  add_executable(neo_fft_dispatch_tests tests/unit/dispatch.cpp)
  target_link_libraries(neo_fft_dispatch_tests PRIVATE neo_fft_core)
  add_test(NAME core.dispatch COMMAND neo_fft_dispatch_tests)
endif()
