cmake_minimum_required(VERSION 3.24)
project(neo_fft VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(NEO_FFT_BUILD_VAPOURSYNTH "Build the VS plugin" ON)
option(NEO_FFT_ENABLE_SIMD "Build SIMD multi-target kernels and runtime dispatch" ON)

# Backward-compatibility aliases for former separate Highway flags
if(DEFINED NEO_FFT_ENABLE_HIGHWAY AND NOT NEO_FFT_ENABLE_HIGHWAY)
  set(NEO_FFT_ENABLE_SIMD OFF CACHE BOOL "Build SIMD multi-target kernels and runtime dispatch" FORCE)
endif()
if(DEFINED NEO_FFT_HWY_SCALAR_ONLY AND NEO_FFT_HWY_SCALAR_ONLY)
  set(NEO_FFT_ENABLE_SIMD OFF CACHE BOOL "Build SIMD multi-target kernels and runtime dispatch" FORCE)
endif()
set(NEO_FFT_ENABLE_HIGHWAY ${NEO_FFT_ENABLE_SIMD})
set(NEO_FFT_HWY_SCALAR_ONLY OFF)
if(NOT NEO_FFT_ENABLE_SIMD)
  set(NEO_FFT_HWY_SCALAR_ONLY ON)
endif()

include(cmake/Dependencies.cmake)

set(NEO_FFT_FFT_TARGETS scalar)
if(NEO_FFT_ENABLE_SIMD)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64|i.86|x86")
    list(APPEND NEO_FFT_FFT_TARGETS sse2 avx2 avx512)
  else()
    list(APPEND NEO_FFT_FFT_TARGETS native)
  endif()
endif()

set(NEO_FFT_FFT_OBJECTS)
foreach(target ${NEO_FFT_FFT_TARGETS})
  add_library(neo_fft_backend_${target} OBJECT src/spectral/fft_backend.cpp)
  target_compile_features(neo_fft_backend_${target} PRIVATE cxx_std_17)
  target_include_directories(neo_fft_backend_${target} PRIVATE src)
  target_include_directories(neo_fft_backend_${target} SYSTEM PRIVATE "${dualsynth2_SOURCE_DIR}/include")
  target_include_directories(neo_fft_backend_${target} SYSTEM PRIVATE "${pocketfft_SOURCE_DIR}")
  set_target_properties(neo_fft_backend_${target} PROPERTIES POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  string(TOUPPER "${target}" _target_upper)
  target_compile_definitions(neo_fft_backend_${target} PRIVATE NEO_FFT_FFT_${_target_upper}=1)
  if(target STREQUAL "scalar")
    target_compile_definitions(neo_fft_backend_${target} PRIVATE POCKETFFT_NO_VECTORS)
    if(MSVC)
      target_compile_options(neo_fft_backend_${target} PRIVATE /fp:strict /W4)
    else()
      target_compile_options(neo_fft_backend_${target} PRIVATE -fno-fast-math -ffp-contract=off -Wall -Wextra -Wpedantic)
    endif()
  elseif(target STREQUAL "sse2")
    if(MSVC)
      target_compile_options(neo_fft_backend_${target} PRIVATE /W4)
    else()
      target_compile_options(neo_fft_backend_${target} PRIVATE -msse2 -Wall -Wextra -Wpedantic)
    endif()
  elseif(target STREQUAL "avx2")
    if(MSVC)
      target_compile_options(neo_fft_backend_${target} PRIVATE /arch:AVX2 /W4)
      if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        target_compile_options(neo_fft_backend_${target} PRIVATE -clang:-ffp-contract=fast)
      else()
        target_compile_options(neo_fft_backend_${target} PRIVATE /fp:fast)
      endif()
    else()
      target_compile_options(neo_fft_backend_${target} PRIVATE -mavx2 -mfma -ffp-contract=fast -Wall -Wextra -Wpedantic)
    endif()
  elseif(target STREQUAL "avx512")
    if(MSVC)
      target_compile_options(neo_fft_backend_${target} PRIVATE /arch:AVX512 /W4)
      if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        target_compile_options(neo_fft_backend_${target} PRIVATE -clang:-ffp-contract=fast)
      else()
        target_compile_options(neo_fft_backend_${target} PRIVATE /fp:fast)
      endif()
    else()
      target_compile_options(neo_fft_backend_${target} PRIVATE -mavx512f -mavx512dq -mavx512cd -mavx512bw -mavx512vl -mfma -ffp-contract=fast -Wall -Wextra -Wpedantic)
    endif()
  endif()
  list(APPEND NEO_FFT_FFT_OBJECTS $<TARGET_OBJECTS:neo_fft_backend_${target}>)
endforeach()

add_library(neo_fft_core STATIC src/spectral/fft.cpp src/algorithms/windows.cpp
  src/algorithms/plan.cpp src/kernels/scalar/spectral.cpp)
target_sources(neo_fft_core PRIVATE ${NEO_FFT_FFT_OBJECTS})
target_compile_definitions(neo_fft_core PRIVATE NEO_FFT_ENABLE_HIGHWAY=$<BOOL:${NEO_FFT_ENABLE_HIGHWAY}>)
target_compile_features(neo_fft_core PUBLIC cxx_std_17)
target_include_directories(neo_fft_core PUBLIC src)
target_include_directories(neo_fft_core SYSTEM PUBLIC "${dualsynth2_SOURCE_DIR}/include")
target_include_directories(neo_fft_core SYSTEM PRIVATE "${pocketfft_SOURCE_DIR}")
if(MSVC)
  target_compile_options(neo_fft_core PUBLIC /fp:strict PRIVATE /W4)
else()
  target_compile_options(neo_fft_core PUBLIC -fno-fast-math -ffp-contract=off PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(neo_fft_core PROPERTIES POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)

if(NEO_FFT_ENABLE_SIMD)
  target_compile_definitions(neo_fft_core PUBLIC NEO_FFT_ENABLE_SIMD=1)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64|i.86|x86")
    target_compile_definitions(neo_fft_core PUBLIC NEO_FFT_FFT_X86_TARGETS=1)
  endif()
endif()

if(NEO_FFT_ENABLE_HIGHWAY)
  foreach(option TESTS EXAMPLES CONTRIB INSTALL)
    set(HWY_ENABLE_${option} OFF CACHE BOOL "" FORCE)
  endforeach()
  set(HWY_FORCE_STATIC_LIBS ON CACHE BOOL "" FORCE)
  FetchContent_Declare(highway
    URL https://github.com/google/highway/archive/refs/tags/1.4.0.tar.gz
    URL_HASH SHA256=e72241ac9524bb653ae52ced768b508045d4438726a303f10181a38f764a453c
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(highway)
  target_sources(neo_fft_core PRIVATE src/kernels/highway/spectral.cpp)
  target_link_libraries(neo_fft_core PUBLIC hwy)
  target_compile_definitions(neo_fft_core PRIVATE HWY_COMPILE_ALL_ATTAINABLE)
  set_target_properties(hwy PROPERTIES POSITION_INDEPENDENT_CODE ON)
  if(NEO_FFT_HWY_SCALAR_ONLY)
    target_compile_definitions(neo_fft_core PRIVATE HWY_COMPILE_ONLY_SCALAR)
  endif()
endif()

include(CTest)
if(NEO_FFT_BUILD_VAPOURSYNTH)
  include(cmake/VapourSynth.cmake)
  add_library(neo_fft MODULE src/plugin/vapoursynth.cpp)
  target_link_libraries(neo_fft PRIVATE neo_fft_core DualSynth::dualsynth)
  target_include_directories(neo_fft SYSTEM PRIVATE "${NEO_FFT_VS_INCLUDE}")
  set_target_properties(neo_fft PROPERTIES PREFIX "" OUTPUT_NAME "neo-fft"
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  if(MINGW)
    target_link_options(neo_fft PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  if(BUILD_TESTING)
    option(NEO_FFT_TEST_VAPOURSYNTH "Run admission tests with an installed Python/VS runtime" ON)
    if(NEO_FFT_TEST_VAPOURSYNTH)
      find_package(Python3 REQUIRED COMPONENTS Interpreter)
      add_test(NAME vs.admission COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/vapoursynth/admission.py" --plugin "$<TARGET_FILE:neo_fft>")
      set_tests_properties(vs.admission PROPERTIES RUN_SERIAL TRUE TIMEOUT 120 FAIL_REGULAR_EXPRESSION "Core freed but")
      add_test(NAME harness.protocol COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/differential/test_protocol.py")
    endif()
  endif()
endif()
if(BUILD_TESTING)
  add_executable(neo_fft_foundation_tests tests/unit/foundation.cpp)
  target_link_libraries(neo_fft_foundation_tests PRIVATE neo_fft_core)
  add_test(NAME core.foundation COMMAND neo_fft_foundation_tests)
  add_executable(neo_fft_algorithm_tests tests/unit/algorithms.cpp)
  target_link_libraries(neo_fft_algorithm_tests PRIVATE neo_fft_core)
  add_test(NAME core.algorithms COMMAND neo_fft_algorithm_tests)
  add_executable(neo_fft_dispatch_tests tests/unit/dispatch.cpp)
  target_link_libraries(neo_fft_dispatch_tests PRIVATE neo_fft_core)
  add_test(NAME core.dispatch COMMAND neo_fft_dispatch_tests)
endif()
